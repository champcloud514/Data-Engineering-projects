# Step 1: Create an Auto-Ingest Snowpipe
Ensure you already have your file format, external stage, and target table created. Then, create the pipe with AUTO_INGEST = TRUE.

CREATE OR REPLACE PIPE my_s3_auto_pipe
  AUTO_INGEST = TRUE
  AS
  COPY INTO my_target_table
  FROM @my_s3_stage
  FILE_FORMAT = (FORMAT_NAME = 'my_csv_format');
